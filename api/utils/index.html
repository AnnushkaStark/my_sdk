    def get_default_logging_config(log_level: LogLevel = "INFO") -> dict[str, Any]:
    """Формирует словарь конфигурации для `logging.config.dictConfig`.

    Args:
        log_level: Минимальный уровень логирования (по умолчанию `"INFO"`).

    Returns:
        Словарь конфигурации логгеров, хендлеров и форматтеров.
    """
    return {
        "version": 1,
        "disable_existing_loggers": False,
        "filters": {
            "context": {"()": ContextFilter},
        },
        "formatters": {
            "app": {
                "()": LogFormatter,
                "format": _APP_FMT,
                "datefmt": _DATE_FMT,
            },
            "system": {
                "()": LogFormatter,
                "format": _SYSTEM_FMT,
                "datefmt": _DATE_FMT,
            },
        },
        "handlers": {
            "stdout_app": {
                "class": "logging.StreamHandler",
                "stream": "ext://sys.stdout",
                "formatter": "app",
                "filters": ["context"],
            },
            "stdout_system": {
                "class": "logging.StreamHandler",
                "stream": "ext://sys.stdout",
                "formatter": "system",
            },
            "stderr": {
                "class": "logging.StreamHandler",
                "stream": "ext://sys.stderr",
                "formatter": "system",
            },
        },
        "root": {
            "level": log_level,
            "handlers": ["stdout_system"],
        },
        "loggers": {
            "app": {
                "level": log_level,
                "handlers": ["stdout_app"],
                "propagate": False,
            },
            "system": {
                "level": log_level,
                "handlers": ["stdout_system"],
                "propagate": False,
            },
            "uvicorn": {"handlers": [], "propagate": True},
            "uvicorn.error": {"handlers": ["stderr"], "propagate": False},
            "uvicorn.access": {"handlers": [], "propagate": False},
            "gunicorn": {"handlers": [], "propagate": True},
            "gunicorn.error": {"handlers": ["stderr"], "propagate": False},
            "gunicorn.access": {"handlers": [], "propagate": False},
        },
    }

    async def domain_error_exception_handler(
    request: Request,
    exc: DomainError,
) -> JSONResponse:
    """Глобальный обработчик исключений `DomainError` для FastAPI.

    Преобразует доменное исключение в JSON-ответ со статус-кодом из
    `ERROR_STATUS_MAP` (или `500`, если код не зарегистрирован).

    Args:
        request: Входящий HTTP-запрос FastAPI.
        exc: Перехваченное доменное исключение.

    Returns:
        JSON-ответ формата `{"detail": "<значение кода>"}`.
    """
    status_code = ERROR_STATUS_MAP.get(exc.code, 500)

    return JSONResponse(
        status_code=status_code,
        content={"detail": exc.code.value},
    )

    def errs(
    **_codes: enum.Enum | Iterable[enum.Enum],
) -> dict[int, dict[str, Any]]:
    """Генерирует словарь `responses` для эндпоинтов FastAPI из кодов ошибок.

    Позволяет декларативно описывать ошибки в роутах через именованные аргументы
    вида `e400=...`, `e404=...`, `e409=...` и автоматически проверяет, что один
    и тот же код ошибки не продублирован в разных HTTP-статусах.

    Args:
        **_codes: Именованные аргументы формата `e<HTTP_STATUS>`, принимающие
            один элемент `enum.Enum` или коллекцию таких элементов.

    Returns:
        Словарь формата `{status_code: {"model": BaseError, "description": ...}}`
        для передачи в параметр `responses` роутера FastAPI.

    Raises:
        RuntimeError: Если один и тот же код ошибки передан более одного раза.
    """
    ret: dict[int, dict[str, Any]] = {}
    processed: set[enum.Enum] = set()

    for status_codes, codes in _codes.items():
        if not isinstance(codes, Iterable):
            codes = (codes,)

        codes_tuple = tuple(codes)
        if set(codes_tuple) & processed:
            raise RuntimeError("Error codes duplicated")

        processed |= set(codes_tuple)
        status_code = int(status_codes.strip("e"))

        ret[status_code] = {
            "model": BaseError,
            "description": _format_description(codes_tuple),
        }

    return ret

    def normalize_phone(value: str, region: str = "RU") -> str:
    """Проверяет номер телефона и приводим его к международному формату E.164.

    Args:
        value: Исходная строка с номером телефона.
        region: Код региона по умолчанию (по умолчанию `"RU"`).

    Returns:
        Нормализованный номер в формате E.164 (например, `"+79991234567"`).

    Raises:
        ValueError: Если строка не является валидным номером телефона.
    """
    try:
        parsed = phonenumbers.parse(value, region)
    except phonenumbers.NumberParseException as exc:
        raise ValueError(f"Invalid phone number: {value}") from exc

    if not phonenumbers.is_valid_number(parsed):
        raise ValueError(f"Invalid phone number: {value}")

    return phonenumbers.format_number(parsed, phonenumbers.PhoneNumberFormat.E164)

    def validate_password(password: str) -> str:
    """Проверяет базовые требования к формату пароля (длина 8–128 символов, без пробелов)."""
    if (
        not isinstance(password, str)
        or not 8 <= len(password) <= 128
        or any(character.isspace() for character in password)
    ):
        raise ValueError("invalid password format")
    return password
