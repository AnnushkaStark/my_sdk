    def build_url(self) -> URL:
    """Формирует безопасный URL подключения для `asyncpg`."""
    return URL.create(
        drivername="postgresql+asyncpg",
        username=self.USER,
        password=self.PASSWORD,
        host=self.HOST,
        port=self.PORT,
        database=self.DB,
    )

    def create_postgres_engine(
    settings: PostgresSettings | None = None,
    pool_size: int = 100,
    max_overflow: int = 20,
    echo: bool = False,
    **engine_kwargs: Any,
) -> AsyncEngine:

    return create_async_engine(
        url=cfg.build_url(),
        pool_size=pool_size,

        echo=echo,
        pool_pre_ping=True,
        **engine_kwargs,
    )

    @asynccontextmanager
async def get_clickhouse_conn(
    settings: ClickHouseSettings | None = None,
    env_prefix: str = "CLICKHOUSE_",
    init_query: str | None = None,
    **client_kwargs: Any,
) -> AsyncIterator[AsyncClient]:
    """Асинхронный контекстный менеджер для работы с клиентом ClickHouse.

    Автоматически читает параметры подключения из переменных окружения,
    выбирает целевую БД, при необходимости выполняет стартовый DDL-скрипт
    `init_query` и гарантированно закрывает соединение при выходе из контекста.

    Args:
        settings: Готовый объект `ClickHouseSettings`. Если не передан, загружается
            автоматически по префиксу `env_prefix`.
        env_prefix: Префикс переменных окружения (по умолчанию `"CLICKHOUSE_"`).
        init_query: Необязательный SQL-запрос инициализации (например, `CREATE TABLE IF NOT EXISTS ...`).
        **client_kwargs: Дополнительные параметры для `clickhouse_connect.get_async_client`.

    Yields:
        Активный асинхронный клиент `AsyncClient`.
    """
    cfg = settings or ClickHouseSettings.from_prefix(env_prefix)
    conn = await clickhouse_connect.get_async_client(
        database=cfg.DB,
        host=cfg.HOST,
        port=cfg.PORT,
        username=cfg.USER,
        password=cfg.PASSWORD,
        **client_kwargs,
    )
    try:
        await conn.command(f"USE {cfg.DB}")
        if init_query:
            await conn.command(init_query)
        yield conn
    finally:
        await conn.close()

    @asynccontextmanager
async def get_redis(
    settings: RedisSettings | None = None,
    env_prefix: str = "REDIS_",
    **redis_kwargs: Any,
) -> AsyncGenerator[Redis, None]:
    """Асинхронный контекстный менеджер для работы с клиентом Redis.

    Автоматически читает `REDIS_URL` (или `<prefix>URL`) из переменных окружения
    и гарантированно закрывает соединение при выходе из блока `async with`.

    Args:
        settings: Готовый объект `RedisSettings`. Если не передан, загружается
            из переменных окружения.
        env_prefix: Префикс переменных окружения (по умолчанию `"REDIS_"`).
        **redis_kwargs: Дополнительные параметры для `Redis.from_url`.

    Yields:
        Активный асинхронный клиент `Redis`.
    """
    cfg = settings or RedisSettings.from_prefix(env_prefix)
    redis_client = Redis.from_url(cfg.URL, **redis_kwargs)
    try:
        yield redis_client
    finally:
        await redis_client.aclose()
