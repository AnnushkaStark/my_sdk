    ObservabilityBuilder 

    ObservabilityBuilder(
    app,
    service_name,
    service_version,
    *,
    excluded_urls=frozenset(),
)

    Билдер для подключения систем наблюдаемости (observability) к приложению FastAPI.

    Каждый метод with_* сразу инициализирует соответствующий бэкенд и возвращает self (или вложенный _OtlpBuilder), позволяя объединять вызовы в цепочку. Все бэкенды опциональны — подключайте только те, что используются в вашей инфраструктуре.

    Example

    Базовая настройка OTLP-трассировки, Prometheus-метрик и профилирования Pyroscope:

    ObservabilityBuilder(
    app,
    service_name="my-service",
    service_version="1.0.0",
    excluded_urls=frozenset({"/health", "/metrics"}),
).with_otlp("alloy:4317")
    .with_logging()
    .with_httpx()
    .with_fastapi()
    .done()
.with_prometheus()
.with_pyroscope("pyroscope")

    Переопределение имени сервиса для отдельного бэкенда:

    ObservabilityBuilder(
    app, service_name="my-service", service_version="1.0.0"
)
    .with_otlp("alloy:4317")
        .with_logging()
        .with_fastapi()
        .done()
    .with_prometheus()
    .with_pyroscope(
        "pyroscope", 4040, service_name="my-service-profiling"
    )

    def with_otlp(
    self,
    endpoint: str,
    *,
    service_name: str | None = None,
    service_version: str | None = None,
    insecure: bool = True,
) -> _OtlpBuilder:
    """Создаёт и регистрирует `TracerProvider` с экспортом трейсов по протоколу OTLP.

    Args:
        endpoint: Адрес OTLP-коллектора или агента (например, `"alloy:4317"`).
        service_name: Переопределение имени сервиса для OTLP (если отличается от базового).
        service_version: Переопределение версии сервиса для OTLP.
        insecure: Использовать ли незашифрованное gRPC-соединение (по умолчанию `True`).

    Returns:
        Вложенный билдер `_OtlpBuilder` для инструментации FastAPI, HTTPX и логов.
    """
    tracer_provider = _create_tracer_provider(
        endpoint=endpoint,
        service_name=service_name or self._service_name,
        service_version=service_version or self._service_version,
        insecure=insecure,
    )
    trace.set_tracer_provider(tracer_provider)
    return _OtlpBuilder(parent=self, app=self._app, tracer_provider=tracer_provider)

    async def dispatch(
    self,
    request: Request,
    call_next: RequestResponseEndpoint,
) -> Response:
    """Обрабатывает входящий HTTP-запрос, логирует метрики и пробрасывает `X-Request-Id`.

    Args:
        request: Входящий объект запроса Starlette/FastAPI.
        call_next: Функция передачи управления следующему обработчику.

    Returns:
        HTTP-ответ `Response` с добавленным заголовком `X-Request-Id`.
    """
    if request.url.path in self._excluded_urls:
        return await call_next(request)

    request_id = request.headers.get(REQUEST_ID_HEADER) or uuid.uuid4().hex
    token = request_id_ctx_var.set(request_id)

    ip = request.headers.get(REAL_IP_HEADER) or (request.client.host if request.client else None) or "unknown"
    method = request.method
    path = request.url.path
    query = request.url.query
    full_url = f"{path}?{query}" if query else path
    user_agent = request.headers.get("user-agent", "-")

    start_time = time.perf_counter()
    try:
        response = await call_next(request)
    finally:
        request_id_ctx_var.reset(token)

    duration_ms = int((time.perf_counter() - start_time) * _SECONDS_TO_MS)
    status_code = response.status_code
    size = int(response.headers.get("content-length") or 0)

    response.headers[REQUEST_ID_HEADER] = request_id

    if status_code >= status.HTTP_500_INTERNAL_SERVER_ERROR:
        self._logger.error(
            _ACCESS_ERROR_FMT,
            ip,
            method,
            full_url,
            status_code,
            duration_ms,
            user_agent,
            size,
        )
    elif status_code >= status.HTTP_400_BAD_REQUEST:
        self._logger.warning(
            _ACCESS_WARNING_FMT,
            ip,
            method,
            full_url,
            status_code,
            duration_ms,
            user_agent,
            size,
        )
    else:
        self._logger.info(
            _ACCESS_INFO_FMT,
            ip,
            method,
            full_url,
            status_code,
            duration_ms,
            user_agent,
            size,
        )

    return response

    async def dispatch(self, request: Request, call_next: RequestResponseEndpoint) -> Response:
    """Обрабатывает HTTP-запрос и обновляет счётчики, гистограммы и гейджи Prometheus.

    Args:
        request: Входящий объект запроса Starlette/FastAPI.
        call_next: Функция передачи управления следующему обработчику.

    Returns:
        Сформированный HTTP-ответ `Response`.
    """
    if request.url.path in self._excluded_urls:
        return await call_next(request)

    if (path := self._get_route_path(request)) is None:
        return await call_next(request)

    service_name = self._service_name
    method = request.method

    REQUESTS.labels(service_name=service_name, method=method, path=path).inc()
    REQUESTS_IN_PROGRESS.labels(service_name=service_name, method=method, path=path).inc()
    before_time = time.perf_counter()
    status_code = HTTP_500_INTERNAL_SERVER_ERROR
    try:
        response = await call_next(request)

    except BaseException as e:
        EXCEPTIONS.labels(
            service_name=service_name,
            method=method,
            path=path,
            exception_type=type(e).__name__,
        ).inc()
        raise

    else:
        status_code = response.status_code
        REQUESTS_PROCESSING_TIME.labels(service_name=service_name, method=method, path=path).observe(
            time.perf_counter() - before_time
        )

    finally:
        RESPONSES.labels(
            service_name=service_name,
            method=method,
            path=path,
            status_code=status_code,
        ).inc()
        REQUESTS_IN_PROGRESS.labels(service_name=service_name, method=method, path=path).dec()

    return response
