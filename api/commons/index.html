    def __init__(
    self,
    settings: S3Settings | None = None,
    env_prefix: str = "S3_",
) -> None:
    """Инициализирует клиент S3-хранилища из переданных настроек или `.env`.

    Args:
        settings: Готовый объект `S3Settings`. Если не передан, загружается
            из переменных окружения по префиксу `env_prefix`.
        env_prefix: Префикс переменных окружения (по умолчанию `"S3_"`).
    """
    self.settings = settings or S3Settings.from_prefix(env_prefix)
    self.session = Session(
        aws_access_key_id=self.settings.ACCESS_KEY,
        aws_secret_access_key=self.settings.ACCESS_SECRET_KEY,
        region_name=self.settings.REGION,
    )
    self.bucket = self.settings.BUCKET
    self.endpoint = self.settings.ENDPOINT
    self.use_ssl = self.settings.USE_SSL

    async def upload_file(
    self,
    file: UploadFile,
    folder: str = "files",
) -> str:
    """Загружает файл FastAPI `UploadFile` в бакет S3.

    Args:
        file: Загружаемый файл из запроса FastAPI.
        folder: Префикс директории внутри бакета (по умолчанию `"files"`).

    Returns:
        Публичный URL сохранённого файла в S3.

    Raises:
        RuntimeError: При ошибке записи объекта в S3.
    """
    async with self._get_client() as client:
        secure_name = self._get_secure_name(
            filename=file.filename,
            folder=folder,
        )
        file_data = await file.read()
        try:
            await client.put_object(
                Bucket=self.bucket,
                Key=secure_name,
                ContentLength=file.size or len(file_data),
                ContentType=file.content_type or "application/octet-stream",
                Body=file_data,
            )
        except Exception as ex:
            raise RuntimeError("Error save file to S3") from ex

        return self._build_public_url(key=secure_name)

    async def remove_file(self, file_url: str) -> str:
    """Удаляет объект из бакета S3 по его публичному URL или ключу.

    Args:
        file_url: Полный URL файла в S3 или ключ объекта внутри бакета.

    Returns:
        Строка `"Ok"` при успешном удалении.

    Raises:
        RuntimeError: При ошибке удаления объекта из S3.
    """
    key = self._extract_key(file_url)
    async with self._get_client() as client:
        try:
            await client.delete_object(Bucket=self.bucket, Key=key)
        except Exception as err:
            raise RuntimeError("Error remove file from S3") from err

        return "Ok"

    async def check_file(
    self,
    obj_picture: str | None = None,
    picture: UploadFile | None = None,
    folder: str = "files",
) -> str | None:
    """Синхронизирует файл при обновлении сущности (загрузка, замена или удаление).

    Args:
        obj_picture: Текущий URL файла, сохранённый в БД (если есть).
        picture: Новый файл из запроса (если передан).
        folder: Префикс директории внутри бакета (по умолчанию `"files"`).

    Returns:
        Новый URL файла после загрузки/замены или `None`, если файл удалён/отсутствует.
    """
    if obj_picture is not None and picture is not None:
        await self.remove_file(obj_picture)
        return await self.upload_file(file=picture, folder=folder)
    if obj_picture is not None and picture is None:
        await self.remove_file(obj_picture)
        return None
    if obj_picture is None and picture is not None:
        return await self.upload_file(file=picture, folder=folder)
    return None

    async def start(self) -> None:
    """Запускает клиент `AIOKafkaProducer` и устанавливает соединение с брокером."""
    if self.producer is not None:
        return

    self.producer = AIOKafkaProducer(
        bootstrap_servers=self.bootstrap_url,
        enable_idempotence=True,
        value_serializer=lambda v: json.dumps(v, ensure_ascii=False).encode("utf-8"),
        **self._producer_kwargs,
    )
    await self.producer.start()

    async def send_message(
    self,
    message: dict[str, Any],
    topic: str,
    key: bytes | str | None = None,
) -> Any:
    """Отправляет JSON-сообщение в указанный топик Kafka и ждёт подтверждения.

    Если продюсер ещё не был запущен явно через `start()`, запускает его
    автоматически перед отправкой.

    Args:
        message: Словарь с полезной нагрузкой события.
        topic: Название целевого топика Kafka.
        key: Необязательный ключ партиционирования (`str` или `bytes`).

    Returns:
        Метаданные отправленной записи (`RecordMetadata`).
    """
    if self.producer is None:
        await self.start()

    assert self.producer is not None
    encoded_key = key.encode("utf-8") if isinstance(key, str) else key
    return await self.producer.send_and_wait(
        topic=topic,
        value=message,
        key=encoded_key,
    )

    async def get_state(self, key: str) -> Any | None:
    """Извлекает и удаляет состояние из Redis с защитой распределённым локом.

    Сначала проверяет наличие значения без блокировки; если ключ ещё не
    появился, захватывает распределённый лок `{key}:lock` и повторяет попытку.

    Args:
        key: Уникальный ключ состояния.

    Returns:
        Десериализованное значение состояния или `None`, если ключ не найден.
    """
    async with get_redis(env_prefix=self.env_prefix) as redis_client:
        if found_obj := await redis_client.get(key):
            await redis_client.delete(key)
            return self._deserialize(found_obj)

        lock_key = f"{key}:lock"
        async with Lock(redis_client, lock_key, timeout=10, blocking_timeout=15):
            if found_obj := await redis_client.get(key):
                await redis_client.delete(key)
                return self._deserialize(found_obj)

    return None

    self,
    recipients: list[EmailStr],
    template: str,
    template_body: dict[str, Any] | None = None,
    subject: str = "",
    attachments: list[UploadFile] | None = None,
) -> dict[str, str]:
    """Рендерит HTML-шаблон письма и отправляет сообщение получателям.

    Args:
        recipients: Список email-адресов получателей.
        template: Имя файла шаблона внутри `templates_dir` (например, `"welcome.html"`).
        template_body: Словарь переменных для подстановки в Jinja2-шаблон.

        RuntimeError: При ошибке рендеринга или отправки письма через SMTP.
    """
    context = template_body or {}
    jinja_template = self._env.get_template(template)
    html = await jinja_template.render_async(subject=subject, **context)

    message = MessageSchema(
        recipients=recipients,
        subject=subject,
        body=html,
        attachments=attachments or [],
        subtype=MessageType.html,
    )

    try:
        await self._fast_mail.send_message(message=message)
    except Exception as exc:
        raise RuntimeError("Error sending email message") from exc

    return {"message": "Message sent successfully"}
